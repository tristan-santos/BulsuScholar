-- BulsuScholar ERD SQL Schema
-- Target database: Supabase PostgreSQL

create extension if not exists "pgcrypto";

-- =========================
-- ENUM TYPES
-- =========================

create type user_role as enum (
	'admin',
	'student',
	'provider'
);

create type account_status as enum (
	'active',
	'pending',
	'archived',
	'inactive'
);

create type application_status as enum (
	'pending',
	'approved',
	'rejected',
	'withdrawn'
);

create type material_type as enum (
	'SOE',
	'Application Form'
);

create type request_status as enum (
	'pending',
	'approved',
	'rejected',
	'downloaded',
	'checked'
);

create type announcement_status as enum (
	'draft',
	'published',
	'archived'
);

-- =========================
-- USERS / AUTH PROFILES
-- =========================

create table public.users (
	id uuid primary key default gen_random_uuid(),
	auth_user_id uuid unique,
	email text not null unique,
	role user_role not null,
	status account_status not null default 'active',
	created_at timestamptz not null default now(),
	updated_at timestamptz not null default now(),
	last_login_at timestamptz
);

-- =========================
-- STUDENTS
-- =========================

create table public.students (
	id uuid primary key default gen_random_uuid(),
	user_id uuid unique references public.users(id) on delete set null,
	student_number text not null unique,
	first_name text not null,
	middle_name text,
	last_name text not null,
	email text not null,
	contact_number text,
	course text,
	year_level text,
	section text,
	address text,
	status account_status not null default 'active',
	archived boolean not null default false,
	scholarship_conflict_warning boolean not null default false,
	scholarship_restriction_reason text,
	scholarship_conflict_message text,
	created_at timestamptz not null default now(),
	updated_at timestamptz not null default now()
);

create table public.pending_students (
	id uuid primary key default gen_random_uuid(),
	student_number text not null unique,
	first_name text not null,
	middle_name text,
	last_name text not null,
	email text not null unique,
	contact_number text,
	course text,
	year_level text,
	section text,
	address text,
	password_hash text,
	status application_status not null default 'pending',
	reviewed_by uuid references public.users(id) on delete set null,
	reviewed_at timestamptz,
	created_at timestamptz not null default now()
);

-- =========================
-- PROVIDERS / GRANTORS
-- =========================

create table public.providers (
	id uuid primary key default gen_random_uuid(),
	user_id uuid unique references public.users(id) on delete set null,
	name text not null,
	provider_type text not null,
	contact_email text,
	contact_number text,
	logo_url text,
	status account_status not null default 'active',
	created_at timestamptz not null default now(),
	updated_at timestamptz not null default now()
);

create table public.grantor_portals (
	id uuid primary key default gen_random_uuid(),
	provider_id uuid not null references public.providers(id) on delete cascade,
	portal_name text not null,
	description text,
	is_enabled boolean not null default true,
	created_at timestamptz not null default now(),
	updated_at timestamptz not null default now()
);

-- =========================
-- SCHOLARSHIPS
-- =========================

create table public.scholarship_programs (
	id uuid primary key default gen_random_uuid(),
	provider_id uuid references public.providers(id) on delete set null,
	name text not null,
	description text,
	total_slots integer,
	status account_status not null default 'active',
	academic_year text,
	created_at timestamptz not null default now(),
	updated_at timestamptz not null default now()
);

create table public.student_scholarships (
	id uuid primary key default gen_random_uuid(),
	student_id uuid not null references public.students(id) on delete cascade,
	scholarship_program_id uuid not null references public.scholarship_programs(id) on delete cascade,
	status application_status not null default 'approved',
	academic_year text,
	is_active boolean not null default true,
	awarded_at timestamptz,
	created_at timestamptz not null default now(),
	unique (student_id, scholarship_program_id, academic_year)
);

create table public.scholarship_applications (
	id uuid primary key default gen_random_uuid(),
	student_id uuid not null references public.students(id) on delete cascade,
	scholarship_program_id uuid not null references public.scholarship_programs(id) on delete cascade,
	status application_status not null default 'pending',
	application_message text,
	submitted_at timestamptz not null default now(),
	reviewed_by uuid references public.users(id) on delete set null,
	reviewed_at timestamptz,
	created_at timestamptz not null default now(),
	updated_at timestamptz not null default now()
);

-- =========================
-- MATERIAL REQUESTS
-- =========================

create table public.soe_requests (
	id uuid primary key default gen_random_uuid(),
	student_id uuid not null references public.students(id) on delete cascade,
	request_number text not null unique,
	material_type material_type not null,
	purpose text,
	status request_status not null default 'pending',
	approved_by uuid references public.users(id) on delete set null,
	approved_at timestamptz,
	rejected_reason text,
	created_at timestamptz not null default now(),
	updated_at timestamptz not null default now()
);

create table public.soe_downloads (
	id uuid primary key default gen_random_uuid(),
	soe_request_id uuid unique references public.soe_requests(id) on delete cascade,
	student_id uuid not null references public.students(id) on delete cascade,
	request_number text not null,
	file_url text,
	checked_status request_status not null default 'downloaded',
	checked_by uuid references public.users(id) on delete set null,
	checked_at timestamptz,
	downloaded_at timestamptz not null default now(),
	created_at timestamptz not null default now()
);

-- =========================
-- ANNOUNCEMENTS
-- =========================

create table public.announcements (
	id uuid primary key default gen_random_uuid(),
	created_by uuid references public.users(id) on delete set null,
	provider_id uuid references public.providers(id) on delete set null,
	title text not null,
	content text not null,
	status announcement_status not null default 'published',
	start_date date,
	end_date date,
	created_at timestamptz not null default now(),
	updated_at timestamptz not null default now()
);

create table public.announcement_attachments (
	id uuid primary key default gen_random_uuid(),
	announcement_id uuid not null references public.announcements(id) on delete cascade,
	file_url text not null,
	file_name text,
	file_type text,
	storage_path text,
	created_at timestamptz not null default now()
);

-- =========================
-- FILE STORAGE METADATA
-- =========================

create table public.uploaded_files (
	id uuid primary key default gen_random_uuid(),
	owner_user_id uuid references public.users(id) on delete set null,
	student_id uuid references public.students(id) on delete cascade,
	provider_id uuid references public.providers(id) on delete cascade,
	bucket_name text not null,
	storage_path text not null,
	file_url text,
	file_name text,
	file_type text,
	file_size bigint,
	created_at timestamptz not null default now()
);

-- =========================
-- INDEXES
-- =========================

create index idx_students_user_id on public.students(user_id);
create index idx_students_student_number on public.students(student_number);
create index idx_students_archived on public.students(archived);
create index idx_pending_students_status on public.pending_students(status);
create index idx_providers_user_id on public.providers(user_id);
create index idx_scholarship_programs_provider_id on public.scholarship_programs(provider_id);
create index idx_student_scholarships_student_id on public.student_scholarships(student_id);
create index idx_student_scholarships_program_id on public.student_scholarships(scholarship_program_id);
create index idx_scholarship_applications_student_id on public.scholarship_applications(student_id);
create index idx_scholarship_applications_program_id on public.scholarship_applications(scholarship_program_id);
create index idx_soe_requests_student_id on public.soe_requests(student_id);
create index idx_soe_requests_status on public.soe_requests(status);
create index idx_soe_downloads_student_id on public.soe_downloads(student_id);
create index idx_announcements_provider_id on public.announcements(provider_id);
create index idx_announcement_attachments_announcement_id on public.announcement_attachments(announcement_id);
create index idx_uploaded_files_owner_user_id on public.uploaded_files(owner_user_id);

-- =========================
-- ERD RELATIONSHIP SUMMARY
-- =========================

-- users 1:1 students
-- users 1:1 providers
-- providers 1:M scholarship_programs
-- providers 1:M grantor_portals
-- students M:M scholarship_programs through student_scholarships
-- students 1:M scholarship_applications
-- scholarship_programs 1:M scholarship_applications
-- students 1:M soe_requests
-- soe_requests 1:0..1 soe_downloads
-- students 1:M soe_downloads
-- announcements 1:M announcement_attachments
-- users/providers/students 1:M uploaded_files
